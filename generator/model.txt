    let getVal = function(name) { let n = u[name]; if (n && n._element) return n._element.value || ""; let w = window.input && window.input[name]; return (w === undefined || w === null) ? "" : String(w); };

    if (!raw || !raw.trim()) return;

    let post = getVal("postPrompt").trim();
    let combined = raw.trim();

    window.input.description = combined;
